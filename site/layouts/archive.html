{{/* A year (or category) archive generated entirely from data files. Set with front matter: layout: archive dataset: boards # a top-level key under site/data peopleDisplay: rows # optional; "cards" (the default) or "rows" peopleDisplay picks the partial each group's members are drawn with. "cards" gives one card per person, for a board or a set of officials: distinct offices, one name in each. "rows" gathers the names under their shared role, for data where many people have the same one, such as the award years in honours.yaml. The choice lives here rather than inside the partials so neither has to know which dataset it is drawing. The dataset is a map from an arbitrary key (a year, a slug) to a group of {heading, sortKey, members}. data/boards and data/officials get one entry per year from one file per year; data/honours.yaml holds every entry in a single file, grouped by category instead of year. Either shape works here because both use the same {heading, sortKey, members} structure. Groups render highest sortKey first, so a year file uses the year itself (newest first) and a hand-grouped file like honours.yaml uses an explicit rank (its first group gets the highest number). */}} {{ define "main" }} {{ $groups := slice }} {{ range index hugo.Data .Params.dataset }} {{ $groups = $groups | append . }} {{ end }} {{ $sorted := sort $groups "sortKey" "desc" }} {{ $display := .Params.peopleDisplay | default "cards" }} {{ $body := slice (string .Content) }} {{/* A jump list, so an archive of sixty years is navigable without scrolling through all of it. */}} {{ if gt (len $sorted) 4 }} {{ $jumps := slice }} {{ range $sorted }} {{ $jumps = $jumps | append (printf `%s` (.heading | urlize) .heading) }} {{ end }} {{ $body = $body | append (printf `
%s
` .Title (delimit $jumps "")) }} {{ end }} {{ range $sorted }} {{ $members := "" }} {{ if eq $display "rows" }} {{ $members = partial "name-rows.html" .members }} {{ else }} {{ $members = partial "people-grid.html" .members }} {{ end }} {{ $body = $body | append (printf `

%s

%s
` (.heading | urlize) .heading $members) }} {{ end }} {{ partial "page-shell.html" (dict "page" . "body" (delimit $body "" | safeHTML)) }} {{ end }}
