{{/* A year (or category) archive generated entirely from data files. Set with front matter: layout: archive dataset: boards # a top-level key under site/data peopleDisplay: rows # optional peopleDisplay picks how each group's members are drawn. The default is a definition list, one row per office: distinct titles with one name in each, as in a board or a set of officials. "rows" gathers the names under their shared role, for data where many people have the same one, such as the award years in honours.yaml. The choice lives here rather than inside name-rows.html so that partial does not have to know which dataset it is drawing. The dataset is a map from an arbitrary key (a year, a slug) to a group of {heading, sortKey, members}. data/boards and data/officials get one entry per year from one file per year; data/honours.yaml holds every entry in a single file, grouped by category instead of year. Either shape works here because both use the same {heading, sortKey, members} structure. Groups render highest sortKey first, so a year file uses the year itself (newest first) and a hand-grouped file like honours.yaml uses an explicit rank (its first group gets the highest number). */}} {{ define "main" }} {{ $groups := slice }} {{ range index hugo.Data .Params.dataset }} {{ $groups = $groups | append . }} {{ end }} {{ $sorted := sort $groups "sortKey" "desc" }} {{ $display := .Params.peopleDisplay }} {{ $isFi := eq .Language.Lang "fi" }} {{ $body := slice (string .Content) }} {{/* A jump list, so an archive of sixty years is navigable without scrolling through all of it. */}} {{ if gt (len $sorted) 4 }} {{ $jumps := slice }} {{ range $sorted }} {{ $jumps = $jumps | append (printf `%s` (.heading | urlize) .heading) }} {{ end }} {{ $body = $body | append (printf `
%s
` .Title (delimit $jumps "")) }} {{ end }} {{ range $sorted }} {{ $members := "" }} {{ if eq $display "rows" }} {{ $members = partial "name-rows.html" .members }} {{ else }} {{/* The archive is read down a column of years, so each year is a list of office and name rather than a grid of faces: the pictures do not exist for most of these decades, and the offices are what one reads an old board for. */}} {{ $rows := slice }} {{ range .members }} {{ $role := .role }} {{ if and (not $isFi) .roleEn }}{{ $role = .roleEn }}{{ end }} {{ $rows = $rows | append (printf `
%s
%s
` (htmlEscape $role) (htmlEscape .name)) }} {{ end }} {{ $members = printf `
%s
` (delimit $rows "") }} {{ end }} {{/* The id sits on the heading, which is what the jump list aims at and what the scroll offset under the sticky header is set on. */}} {{ $body = $body | append (printf `

` (.heading | urlize) .heading $members) }} {{ end }} {{ partial "page-shell.html" (dict "page" . "body" (delimit $body "" | safeHTML)) }} {{ end }}
